Turns one token into its value for moment, or returns it as literal text.

This powers format. You generally won't call it directly, but it's public so you can extend or override token behaviour. Missing values on moment are treated as 0 (or 1 for the date and month).

Implementation

String formatToken(Moment moment, String token) {
  final int second = moment.second ?? 0;
  final int minute = moment.minute ?? 0;
  final int hour = moment.hour ?? 0;
  final int date = moment.date ?? 1;
  final int month = moment.month ?? 1;
  final int year = moment.year ?? 0;
  final int hour12 = hour % 12 == 0 ? 12 : hour % 12;

  switch (token) {
    case ddd:
      return (_WEEK_DAY_NAMES[moment.weekday] ?? '   ').substring(0, 3);
    case dddd:
      return _WEEK_DAY_NAMES[moment.weekday] ?? '';
    case D:
      return '$date';
    case Do:
      return '$date${_getOrdinal(date)}';
    case DD:
      return _pad(date);
    case M:
      return '$month';
    case Mo:
      return '$month${_getOrdinal(month)}';
    case MM:
      return _pad(month);
    case MMM:
      return _MONTH_NAMES[month]!.substring(0, 3);
    case MMMM:
      return _MONTH_NAMES[month]!;
    case yy:
      return _pad(year % 100);
    case yyyy:
      return '$year';
    case h:
      return '$hour12';
    case hh:
      return _pad(hour12);
    case H:
      return '$hour';
    case HH:
      return _pad(hour);
    case m:
      return '$minute';
    case mm:
      return _pad(minute);
    case s:
      return '$second';
    case ss:
      return _pad(second);
    case SSS:
      return '${moment.millisecond ?? 0}'.padLeft(3, '0');
    case SSSSSS:
      return '${moment._subSecondMicroseconds}'.padLeft(6, '0');
    case A:
      return moment.clockPhase;
    case a:
      return moment.clockPhase.toLowerCase();
    case t:
      return moment.time;
    case T:
      return moment.clock;
    case dateNumeric:
      return '${_pad(date)}/${_pad(month)}/$year';
    default:
      if (token.length >= 2 && token.startsWith('[') && token.endsWith(']')) {
        return token.substring(1, token.length - 1);
      }
      return token;
  }
}
