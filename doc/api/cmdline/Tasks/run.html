Runs tasks, up to concurrency at a time, showing each one's state: waiting, spinning, ticked or crossed, with its status and progress.

A task that throws is marked failed and the others carry on. Gives back a report for each task, in order. When output is piped, one line is printed per task as it ends.

final reports = await Tasks.run([
  Task('Download', (task) async {
    for (var i = 1; i <= 10; i++) {
      await fetchChunk(i);
      task.progress(i, 10);
    }
  }),
  Task('Build', (task) async {
    task.update('compiling');
    await build();
  }),
]);
if (reports.any((r) => !r.ok)) exitCode = 1;

static Future<List<TaskReport>> run(
  List<Task> tasks, {
  int concurrency = 4,
  List<String> frames = kSPINNER_FRAMES,
}) async {
  final terminal = Terminal.current;
  final live = terminal.supportsAnsi;
  final region = LiveRegion(terminal);
  final states = List.filled(tasks.length, _State.waiting);
  final clocks = [for (final _ in tasks) Stopwatch()];
  final errors = List<Object?>.filled(tasks.length, null);
  var frame = 0;
  late final List<TaskContext> contexts;

  void draw() {
    if (!live) return;
    region.draw([
      for (var i = 0; i < tasks.length; i++)
        _line(
          tasks[i].label,
          states[i],
          contexts[i],
          clocks[i].elapsed,
          frames[frame % frames.length],
        ).fitWidth(terminal.columns - 1),
    ]);
  }

  void log(String line) {
    if (live) region.clear();
    PromptRunner.say(line, terminal);
    draw();
  }

  contexts = [for (final _ in tasks) TaskContext._(draw, log)];
  final timer = live
      ? Timer.periodic(const Duration(milliseconds: 80), (_) {
          frame++;
          draw();
        })
      : null;

  var next = 0;
  Future<void> worker() async {
    while (next < tasks.length) {
      final i = next++;
      states[i] = _State.running;
      clocks[i].start();
      draw();
      try {
        await tasks[i].work(contexts[i]);
        states[i] = _State.passed;
      } catch (error) {
        states[i] = _State.failed;
        errors[i] = error;
        contexts[i].status = '$error';
      }
      clocks[i].stop();
      if (live) {
        draw();
      } else {
        PromptRunner.say(
          _line(
            tasks[i].label,
            states[i],
            contexts[i],
            clocks[i].elapsed,
            '',
          ),
          terminal,
        );
      }
    }
  }

  try {
    await Future.wait([
      for (
        var w = 0;
        w < math.max(1, math.min(concurrency, tasks.length));
        w++
      )
        worker(),
    ]);
  } finally {
    timer?.cancel();
    if (live) {
      region.finish([
        for (var i = 0; i < tasks.length; i++)
          _line(
            tasks[i].label,
            states[i],
            contexts[i],
            clocks[i].elapsed,
            '',
          ),
      ]);
    }
  }

  return [
    for (var i = 0; i < tasks.length; i++)
      TaskReport(tasks[i].label, errors[i], clocks[i].elapsed),
  ];
}
