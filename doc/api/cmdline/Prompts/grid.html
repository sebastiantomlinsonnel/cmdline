static T grid<T>(
  String prompt,
  List<T> options, {
  String Function(T option)? format,
  T? defaultOption,
  int? columns,
}) {
  final start = defaultOption == null ? -1 : options.indexOf(defaultOption);

  return PromptRunner.run(
    GridPrompt<T>(
      prompt,
      options,
      format: format,
      columns: columns,
      start: start < 0 ? 0 : start,
      hasDefault: start >= 0,
    ),
  );
}
